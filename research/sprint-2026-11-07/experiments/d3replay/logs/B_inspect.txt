def raw_aes(circuit):
    assert circuit['n_inputs'] == 32
    values = [1 << i for i in range(32)]
    depths = [0]*32
    births = {}
    for index, (a, b) in enumerate(circuit['gates']):
        assert 0 <= a < 32+index and 0 <= b < 32+index
        v = values[a] ^ values[b]
        assert v and v not in values
        births[v] = (values[a], values[b])
        values.append(v)
        depths.append(1+max(depths[a], depths[b]))
    actual = [values[i] for i in circuit['outputs']]
    assert len(actual) == 32 and max(depths) == 3
    expected = [0]*32
    def twice(x):
        return ((x << 1) ^ (0x1b if x & 0x80 else 0)) & 255
    for source in range(32):
        column = [0]*4
        column[source//8] = 1 << (source%8)
        transformed = []
        for row in range(4):
            a, b, c, d = (column[(row+j)%4] for j in range(4))
            transformed.append(twice(a) ^ twice(b) ^ b ^ c ^ d)
        for row, value in enumerate(transformed):
            for bit in range(8):
                if value >> bit & 1:
                    expected[8*row+bit] |= 1 << source
    assert actual == expected
    outputs = set(actual)
    assert not outputs & {p for ab in births.values() for p in ab}
    for v, (a, b) in births.items():
        if v in outputs:
            continue
        assert not a & b
        assert sorted((a.bit_count(), b.bit_count())) == {2: [1, 1], 3: [1, 2], 4: [2, 2]}[v.bit_count()]

def choices(h):
    """Enumerate supports independently by pair subset and XOR remainder."""
    bits=[1<<i for i in range(h.bit_length()) if h>>i&1]
    if len(bits)==1:return [()]
    if len(bits)==2:return [(h,)]
    out=set()
    for a,b in combinations(bits,2):
        p=a|b;rest=h^p
        if rest.bit_count()==1:out.add((p,))
        elif rest.bit_count()==2:out.add(tuple(sorted((p,rest))))
    assert len(bits) in (3,4) and len(out)==3
    return sorted(out)


def reconstruct(circuit):

    parents={p for target in outputs for p in births[target]}
    forced={p for p in parents if p.bit_count()==2}

    universes={h:{p for option in opts for p in option} for h,opts in requirements.items()}

    # This differs from the producer's clique graph over candidate-pair vertices.
